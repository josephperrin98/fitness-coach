```python
"""Fitness as one number, VDOT, and the paces that follow from it (D29, D30).

VDOT is Daniels and Gilbert's model: the oxygen cost of running at a speed,
divided by the fraction of VO2max a runner can hold for that duration. One race
or test result gives a VDOT; every training pace follows from it. Standard
library only: the model is two published equations.
"""
from __future__ import annotations

import math

RACE_METRES = {"half": 21097.5, "marathon": 42195.0}


def _oxygen_cost(metres_per_min: float) -> float:
    return -4.60 + 0.182258 * metres_per_min + 0.000104 * metres_per_min ** 2


def _fraction_held(minutes: float) -> float:
    return 0.8 + 0.1894393 * math.exp(-0.012778 * minutes) + 0.2989558 * math.exp(-0.1932605 * minutes)


def _speed_for_cost(cost: float) -> float:
    # Inverse of _oxygen_cost: the positive root of a*v^2 + b*v + c = 0.
    a, b, c = 0.000104, 0.182258, -4.60 - cost
    return (-b + math.sqrt(b * b - 4 * a * c)) / (2 * a)


def vdot(distance_m: float, seconds: float) -> float:
    minutes = seconds / 60
    return _oxygen_cost(distance_m / minutes) / _fraction_held(minutes)


def race_seconds(vdot_value: float, distance_m: float) -> float:
    """The time a runner with this VDOT is predicted to run the distance in."""
    lo, hi = 60.0, 50_000.0
    for _ in range(100):   # bisection: vdot() falls as the time grows
        mid = (lo + hi) / 2
        lo, hi = (mid, hi) if vdot(distance_m, mid) > vdot_value else (lo, mid)
    return (lo + hi) / 2


def pace_band(vdot_value: float, kind: str, intensity: dict, race_m: float | None = None) -> tuple[int, int]:
    """(fast, slow) pace in seconds per km for a session kind."""
    if kind == "race":
        if race_m is None:
            raise ValueError("race pace needs a race distance")
        pace = race_seconds(vdot_value, race_m) / (race_m / 1000)
        return round(pace / 1.01), round(pace / 0.99)
    low, high = intensity[kind]
    return round(60_000 / _speed_for_cost(high * vdot_value)), round(60_000 / _speed_for_cost(low * vdot_value))


def fmt_pace(sec_per_km: float) -> str:
    s = round(sec_per_km)
    return f"{s // 60}:{s % 60:02d}"
```

```python
def vdot_line(weeks: Sequence[Week], start_day: date, start_vdot: float,
              target_day: date, target_vdot: float) -> dict[date, float]:
    """Planned VDOT for each week from the checkpoint's week on (D30).

    It rises in equal steps in loading weeks up to the target's week and holds in
    recovery, taper and race weeks, so recovery is never asked to make you fitter.
    """
    first = start_day - timedelta(days=start_day.weekday())
    last = target_day - timedelta(days=target_day.weekday())
    ahead = [w for w in weeks if w.start >= first]
    rising = [w for w in ahead if first < w.start <= last and w.type in LOADING_TYPES]
    step = (target_vdot - start_vdot) / len(rising) if rising and target_vdot > start_vdot else 0.0
    line, value = {}, start_vdot
    for w in ahead:
        if w in rising:
            value += step
        line[w.start] = round(value, 2)
    return line


def pace_vdot(line_vdot: float, checkpoint_vdot: float, cap: float) -> float:
    """The VDOT paces are computed from: the planned line, never more than cap above the last test."""
    return min(line_vdot, checkpoint_vdot + cap)
```

```python
"""Checkpoints: test results that set the athlete's measured VDOT (D31).

A result usually comes from Strava, and Strava data is never stored (D21). So the
agent shows the numbers and records them only after the athlete confirms them: what
is stored is the athlete's own statement of their result. VDOT is derived on read,
never stored, so a change to the formula can never leave stale numbers behind.
"""
from __future__ import annotations

import sqlite3
from dataclasses import dataclass
from datetime import date, datetime, timezone

from .fitness import vdot

CHECKPOINT_KINDS = ("6min", "5k", "race")

SCHEMA = """
CREATE TABLE IF NOT EXISTS checkpoints (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    day         TEXT NOT NULL,
    kind        TEXT NOT NULL,
    distance_m  REAL NOT NULL,
    seconds     REAL NOT NULL,
    created_at  TEXT NOT NULL
);
"""


@dataclass(frozen=True)
class Checkpoint:
    day: date
    kind: str
    distance_m: float
    seconds: float

    def __post_init__(self) -> None:
        if self.kind not in CHECKPOINT_KINDS:
            raise ValueError(f"unknown checkpoint kind {self.kind!r}; expected one of {CHECKPOINT_KINDS}")
        if self.distance_m <= 0 or self.seconds <= 0:
            raise ValueError("distance and time must be positive")

    @property
    def vdot(self) -> float:
        return vdot(self.distance_m, self.seconds)


class CheckpointStore:
    def __init__(self, path: str = ":memory:") -> None:
        # Same threading note as CheckInStore: one user, calls arrive one at a time.
        self.db = sqlite3.connect(path, check_same_thread=False)
        self.db.row_factory = sqlite3.Row
        self.db.executescript(SCHEMA)

    def record(self, c: Checkpoint) -> Checkpoint:
        self.db.execute(
            "INSERT INTO checkpoints (day, kind, distance_m, seconds, created_at) VALUES (?,?,?,?,?)",
            (c.day.isoformat(), c.kind, c.distance_m, c.seconds,
             datetime.now(timezone.utc).isoformat(timespec="seconds")))
        self.db.commit()
        return c

    def latest(self, as_of: date) -> Checkpoint | None:
        row = self.db.execute("SELECT * FROM checkpoints WHERE day <= ? ORDER BY day DESC, id DESC LIMIT 1",
                              (as_of.isoformat(),)).fetchone()
        return Checkpoint(day=date.fromisoformat(row["day"]), kind=row["kind"],
                          distance_m=row["distance_m"], seconds=row["seconds"]) if row else None
```

    def paces(self, day: date) -> dict[str, Any]:
        """Pace bands for the week of `day` (D29, D30)."""
        cp = self.checkpoints.latest(day)
        if cp is None:
            return {"error": "no checkpoint recorded yet: record a confirmed test result first"}
        plan, prog = self.plan(), self.rules.paces["progression"]
        race_m = RACE_METRES[plan.goal] if plan else None
        line = cp.vdot
        if plan and plan.target_time:
            target_day = plan.race_date - timedelta(days=prog["target_lead_days"])
            monday = day - timedelta(days=day.weekday())
            line = vdot_line(plan.weeks, cp.day, cp.vdot, target_day,
                             vdot(race_m, plan.target_time)).get(monday, cp.vdot)
        used = pace_vdot(line, cp.vdot, prog["cap_above_checkpoint"])
        kinds = ["easy", "long", "threshold", "interval"] + (["race"] if race_m else [])
        intensity = self.rules.paces["intensity"]
        bands = {k: pace_band(used, k, intensity, race_m) for k in kinds}
        return {
            "day": day.isoformat(),
            "checkpoint": {"day": cp.day.isoformat(), "kind": cp.kind, "vdot": round(cp.vdot, 1)},
            "line_vdot": round(line, 1),
            "vdot_used": round(used, 1),
            "capped": used < line,
            "paces": {k: f"{fmt_pace(fast)}-{fmt_pace(slow)}/km" for k, (fast, slow) in bands.items()},
        }
```

@mcp.tool(annotations=READ)
def get_paces(day: str = "") -> str:
    """Pace bands (min:s per km) for easy, long, threshold, interval and race-pace running in the week of 'day' (YYYY-MM-DD, empty = today), from the last confirmed test and the planned VDOT line. 'capped' is true when the plan is ahead of the last test and paces were held back."""
    return _json(coach.paces(_day(day)))


@mcp.tool()
def record_checkpoint(day: str, kind: str, distance_m: float, seconds: float) -> str:
    """Save a test result AFTER the user has confirmed the numbers in the conversation. kind is '6min' (distance covered in the 6-minute test), '5k' (time trial) or 'race'. distance_m in metres, seconds is the effort's duration. Show the user the distance, time and resulting VDOT first; never save a result they have not confirmed."""
    try:
        c = Checkpoint(day=_day(day), kind=kind, distance_m=distance_m, seconds=seconds)
        return _json({"saved": coach.record_checkpoint(c)})
    except ValueError as e:
        return _json({"refused": str(e)})
```
